cmake_minimum_required(VERSION 3.16)
project(MiniKV VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(minikv src/engine.cpp src/replication.cpp)
target_include_directories(minikv PUBLIC include)
if(MSVC)
  target_compile_options(minikv PRIVATE /W4)
else()
  target_compile_options(minikv PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(minikv_cli src/main.cpp)
target_link_libraries(minikv_cli PRIVATE minikv)
add_executable(minikv_tests tests/test_engine.cpp)
target_link_libraries(minikv_tests PRIVATE minikv)
add_executable(minikv_bench benchmarks/benchmark.cpp)
target_link_libraries(minikv_bench PRIVATE minikv)
enable_testing()
add_test(NAME minikv_tests COMMAND minikv_tests)
